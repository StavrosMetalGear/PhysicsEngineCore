cmake_minimum_required(VERSION 3.15)
project(PhysicsEngineDeprecationChecks LANGUAGES C CXX)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_TRY_COMPILE_TARGET_TYPE STATIC_LIBRARY)

include(CheckCSourceCompiles)
include(CheckCXXSourceCompiles)
set(CMAKE_REQUIRED_INCLUDES "${CMAKE_CURRENT_LIST_DIR}/../..")

if(MSVC)
	set(CMAKE_REQUIRED_FLAGS "/we4996")
elseif(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang" AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
	set(CMAKE_REQUIRED_FLAGS "-Werror=deprecated-declarations")
else()
	message(FATAL_ERROR "Deprecation checks require MSVC, GCC, or Clang.")
endif()

foreach(language IN ITEMS C CXX)
	foreach(mode IN ITEMS STATIC SHARED_EXPORT SHARED_IMPORT)
		if(mode STREQUAL "SHARED_EXPORT")
			set(CMAKE_REQUIRED_DEFINITIONS -DPHYSICS_ENGINE_SHARED -DPHYSICS_ENGINE_EXPORTS)
		elseif(mode STREQUAL "SHARED_IMPORT")
			set(CMAKE_REQUIRED_DEFINITIONS -DPHYSICS_ENGINE_SHARED)
		else()
			set(CMAKE_REQUIRED_DEFINITIONS "")
		endif()

		set(source_prefix "#include <PhysicsEngine_export.h>\nPHYSENG_DEPRECATED(\"compatibility test\") PHYSENG_API int pe_deprecation_probe(void);\n")
		set(unused_result "${language}_${mode}_UNUSED_COMPILES")
		set(used_result "${language}_${mode}_USED_COMPILES")
		unset(${unused_result} CACHE)
		unset(${used_result} CACHE)
		if(language STREQUAL "C")
			check_c_source_compiles("${source_prefix}int main(void) { return 0; }" ${unused_result})
			check_c_source_compiles("${source_prefix}int main(void) { return pe_deprecation_probe(); }" ${used_result})
		else()
			check_cxx_source_compiles("${source_prefix}int main(void) { return 0; }" ${unused_result})
			check_cxx_source_compiles("${source_prefix}int main(void) { return pe_deprecation_probe(); }" ${used_result})
		endif()
		if(NOT ${unused_result})
			message(FATAL_ERROR "${language} ${mode}: unused annotated declaration did not compile.")
		endif()
		if(${used_result})
			message(FATAL_ERROR "${language} ${mode}: deprecated use did not produce the expected diagnostic.")
		endif()
	endforeach()
endforeach()

message(STATUS "C and C++ deprecation diagnostics verified for static, shared-export, and shared-import annotations.")
