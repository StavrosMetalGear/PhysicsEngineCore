cmake_minimum_required(VERSION 3.15)
project(PhysicsEngineQuantumLayoutChecks LANGUAGES C CXX)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_TRY_COMPILE_TARGET_TYPE STATIC_LIBRARY)

if(NOT CMAKE_C_COMPILER_ID STREQUAL "MSVC" OR NOT CMAKE_C_COMPILER_ARCHITECTURE_ID STREQUAL "x64" OR
   NOT CMAKE_CXX_COMPILER_ID STREQUAL "MSVC" OR NOT CMAKE_CXX_COMPILER_ARCHITECTURE_ID STREQUAL "x64")
	message(FATAL_ERROR "The Quantum layout snapshot requires MSVC x64 for C and C++.")
endif()

include(CheckCSourceCompiles)
include(CheckCXXSourceCompiles)
get_filename_component(project_root "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(PHYSICSCORE_REPOS_DIR "${project_root}/.." CACHE PATH "Directory containing the sibling repositories")
get_filename_component(quantum_dir "${PHYSICSCORE_REPOS_DIR}/QuantumCore" ABSOLUTE)
if(NOT EXISTS "${quantum_dir}/quantum_c_api.h")
	message(FATAL_ERROR "Quantum API header not found in ${quantum_dir}.")
endif()

file(READ "${project_root}/tests/baselines/quantum_records_msvc_x64.h" snapshot)
set(original "FIELD(QP_EnergyLevel3D, energy, 16, double)")
set(changed "FIELD(QP_EnergyLevel3D, energy, 8, double)")
string(REPLACE "${original}" "${changed}" modified_snapshot "${snapshot}")
if(modified_snapshot STREQUAL snapshot)
	message(FATAL_ERROR "Could not construct the negative layout fixture.")
endif()
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/negative/baselines")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/negative/baselines/quantum_records_msvc_x64.h" "${modified_snapshot}")

foreach(language IN ITEMS C CXX)
	if(language STREQUAL "C")
		file(READ "${project_root}/tests/test_quantum_record_layout_c.c" source)
	else()
		file(READ "${project_root}/tests/test_quantum_record_layout_cpp.cpp" source)
	endif()
	set(positive "${language}_QUANTUM_LAYOUT_COMPILES")
	set(negative "${language}_CHANGED_QUANTUM_LAYOUT_COMPILES")
	unset(${positive} CACHE)
	unset(${negative} CACHE)

	set(CMAKE_REQUIRED_INCLUDES "${quantum_dir};${project_root}/tests")
	if(language STREQUAL "C")
		check_c_source_compiles("${source}" ${positive})
	else()
		check_cxx_source_compiles("${source}" ${positive})
	endif()
	if(NOT ${positive})
		message(FATAL_ERROR "${language}: the Quantum layout snapshot did not compile.")
	endif()

	set(CMAKE_REQUIRED_INCLUDES "${quantum_dir};${CMAKE_CURRENT_BINARY_DIR}/negative;${project_root}/tests")
	if(language STREQUAL "C")
		check_c_source_compiles("${source}" ${negative})
	else()
		check_cxx_source_compiles("${source}" ${negative})
	endif()
	if(${negative})
		message(FATAL_ERROR "${language}: a changed Quantum field offset was not detected.")
	endif()
endforeach()

message(STATUS "C and C++ Quantum layout checks detect the changed fixture.")
