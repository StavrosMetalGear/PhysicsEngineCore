cmake_minimum_required(VERSION 3.15)
project(PhysicsEngineCapabilityBaselineChecks LANGUAGES C CXX)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_TRY_COMPILE_TARGET_TYPE STATIC_LIBRARY)

include(CheckCSourceCompiles)
include(CheckCXXSourceCompiles)
get_filename_component(project_root "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
file(READ "${project_root}/tests/baselines/capability_snapshot.h" snapshot)
set(original "X(PE_CAPABILITY_NUCLEAR, UINT64_C(0x080))")
set(changed "X(PE_CAPABILITY_NUCLEAR, UINT64_C(0x040))")
string(REPLACE "${original}" "${changed}" modified_snapshot "${snapshot}")
if(modified_snapshot STREQUAL snapshot)
	message(FATAL_ERROR "Could not construct the negative capability fixture.")
endif()
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/negative/baselines")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/negative/baselines/capability_snapshot.h" "${modified_snapshot}")

foreach(language IN ITEMS C CXX)
	if(language STREQUAL "C")
		file(READ "${project_root}/tests/test_capability_baseline.c" source)
	else()
		file(READ "${project_root}/tests/test_capability_baseline.cpp" source)
	endif()
	set(positive "${language}_CAPABILITY_BASELINE_COMPILES")
	set(negative "${language}_CHANGED_CAPABILITY_COMPILES")
	unset(${positive} CACHE)
	unset(${negative} CACHE)

	set(CMAKE_REQUIRED_INCLUDES "${project_root};${project_root}/tests")
	if(language STREQUAL "C")
		check_c_source_compiles("${source}" ${positive})
	else()
		check_cxx_source_compiles("${source}" ${positive})
	endif()
	if(NOT ${positive})
		message(FATAL_ERROR "${language}: the capability snapshot did not compile.")
	endif()

	set(CMAKE_REQUIRED_INCLUDES "${project_root};${CMAKE_CURRENT_BINARY_DIR}/negative;${project_root}/tests")
	if(language STREQUAL "C")
		check_c_source_compiles("${source}" ${negative})
	else()
		check_cxx_source_compiles("${source}" ${negative})
	endif()
	if(${negative})
		message(FATAL_ERROR "${language}: a changed capability value was not detected.")
	endif()
endforeach()

message(STATUS "C and C++ capability checks detect the changed fixture.")
