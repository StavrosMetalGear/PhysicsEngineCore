cmake_minimum_required(VERSION 3.15)
project(PhysicsEngineInstalledHeaderChecks LANGUAGES C CXX)

find_package(PhysicsEngine CONFIG REQUIRED)

include("${CMAKE_CURRENT_LIST_DIR}/header_manifest.cmake")
get_target_property(package_includes PhysicsEngine::PhysicsEngine INTERFACE_INCLUDE_DIRECTORIES)
unset(installed_metadata_header CACHE)
find_file(installed_metadata_header NAMES PhysicsMetadata.h PATHS ${package_includes} NO_DEFAULT_PATH)
if(NOT installed_metadata_header)
	message(FATAL_ERROR "PhysicsMetadata.h not found in the installed package include directories.")
endif()
get_filename_component(HEADER_ROOT "${installed_metadata_header}" DIRECTORY)
set(MANIFEST_FILE "${CMAKE_CURRENT_LIST_DIR}/header_manifest.cmake")
include("${CMAKE_CURRENT_LIST_DIR}/verify_manifest.cmake")

set(c_sources)
set(cxx_sources)
foreach(header IN LISTS c_headers)
	string(REPLACE "." "_" source_name "${header}")
	set(c_source "${CMAKE_CURRENT_BINARY_DIR}/${source_name}.c")
	set(cxx_source "${CMAKE_CURRENT_BINARY_DIR}/${source_name}.cpp")
	file(WRITE "${c_source}" "#include <${header}>\n")
	file(WRITE "${cxx_source}" "#include <${header}>\n")
	list(APPEND c_sources "${c_source}")
	list(APPEND cxx_sources "${cxx_source}")
endforeach()

add_library(installed_headers_c OBJECT ${c_sources})
set_target_properties(installed_headers_c PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
target_link_libraries(installed_headers_c PRIVATE PhysicsEngine::PhysicsEngine)

add_library(installed_headers_cpp OBJECT ${cxx_sources})
set_target_properties(installed_headers_cpp PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON)
target_link_libraries(installed_headers_cpp PRIVATE PhysicsEngine::PhysicsEngine)

set(collision_sources)
foreach(header IN LISTS collision_headers)
	string(REPLACE "/" "_" source_name "${header}")
	string(REPLACE "." "_" source_name "${source_name}")
	set(source "${CMAKE_CURRENT_BINARY_DIR}/${source_name}.cpp")
	file(WRITE "${source}" "#include <${header}>\n")
	list(APPEND collision_sources "${source}")
endforeach()
add_library(installed_collision_headers_cpp OBJECT ${collision_sources})
set_target_properties(installed_collision_headers_cpp PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON)
target_link_libraries(installed_collision_headers_cpp PRIVATE PhysicsEngine::PhysicsEngine)
