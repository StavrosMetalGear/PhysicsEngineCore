cmake_minimum_required(VERSION 3.15)
project(PhysicsEnginePackageVersionProbe LANGUAGES CXX)

if(NOT DEFINED REQUEST_VERSION OR REQUEST_VERSION STREQUAL "")
	message(FATAL_ERROR "REQUEST_VERSION is required.")
endif()
if(NOT DEFINED PACKAGE_PREFIX OR NOT IS_DIRECTORY "${PACKAGE_PREFIX}")
	message(FATAL_ERROR "PACKAGE_PREFIX must identify the installed package prefix.")
endif()
if(EXACT_REQUEST)
	find_package(PhysicsEngine ${REQUEST_VERSION} EXACT CONFIG REQUIRED PATHS "${PACKAGE_PREFIX}" NO_DEFAULT_PATH)
else()
	find_package(PhysicsEngine ${REQUEST_VERSION} CONFIG REQUIRED PATHS "${PACKAGE_PREFIX}" NO_DEFAULT_PATH)
endif()
if(NOT TARGET PhysicsEngine::PhysicsEngine)
	message(FATAL_ERROR "The installed PhysicsEngine target is missing.")
endif()
message(STATUS "Found PhysicsEngine package version ${PhysicsEngine_VERSION}.")
